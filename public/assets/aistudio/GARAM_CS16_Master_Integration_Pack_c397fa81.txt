export function deepFreeze<T>(obj: T): Readonly<T> {
  Object.freeze(obj);
  for (const key of Object.getOwnPropertyNames(obj)) {
    const val = (obj as Record<string, unknown>)[key];
    if (val !== null && (typeof val === 'object' || typeof val === 'function') && !Object.isFrozen(val)) {
      deepFreeze(val);
    }
  }
  return obj;
}

  for (let i = 0; i < ladders.length; i++) {
    const l = ladders[i];
    if (
      px >= l.minX - margin &&
      px <= l.maxX + margin &&
      pz >= l.minZ - margin &&
      pz <= l.maxZ + margin &&
      py + KINEMATICS.CAPSULE_HEIGHT_STAND >= l.minY &&
      py <= l.maxY + 0.15
    ) {
      return l;
    }
  }
  return null;
}

/**
 * CS 1.6 func_ladder climbing physics:
 * - Zero gravity while mounted on ladder rungs
 * - Looking up while pressing W climbs UP; looking down while pressing W climbs DOWN
 * - S reverses climb direction; A/D strafes horizontally along the ladder face
 * - Space kicks off perpendicular to ladder normal + upward impulse
 */
export function stepLadderMovement(
  state: PlayerState,
  input: PlayerInput,
  ladder: LadderVolume,
  dt: number,
  colliders: AABBCollider[],
  bounds: MapBounds
): boolean {
  const climbSpeed = 4.80; // ~190 HL units/s
  const fwdX = -Math.sin(input.yaw) * Math.cos(input.pitch);
  const fwdY = Math.sin(input.pitch);
  const fwdZ = -Math.cos(input.yaw) * Math.cos(input.pitch);
  const rightX = Math.cos(input.yaw);
  const rightZ = -Math.sin(input.yaw);

  if (input.jump) {
    state.vx = ladder.normalX * 5.20;
    state.vz = ladder.normalZ * 5.20;
    state.vy = 4.4;
    state.isGrounded = false;
    return false;
  }

  let wishMoveFwd = 0;
  if (input.forward) wishMoveFwd += 1;
  if (input.backward) wishMoveFwd -= 1;

  let wishSide = 0;
  if (input.right) wishSide += 1;
  if (input.left) wishSide -= 1;

  const slowFactor = input.walk ? 0.52 : 1.0;

  // Vertical climb driven by view pitch + forward/back keys (authentic GoldSrc ladder feel)
  const pitchFactor = Math.abs(fwdY) > 0.18 ? fwdY : fwdY >= 0 ? 0.85 : -0.85;
  const climbVy = wishMoveFwd * pitchFactor * climbSpeed * slowFactor;

  // Lateral shimmy along ladder plane
  const tangX = -ladder.normalZ;
  const tangZ = ladder.normalX;
  const sideSpeed = (wishSide * (rightX * tangX + rightZ * tangZ) + wishMoveFwd * (fwdX * tangX + fwdZ * tangZ)) * climbSpeed * 0.65 * slowFactor;

  state.vx = tangX * sideSpeed - ladder.normalX * 0.45 * wishMoveFwd;
  state.vz = tangZ * sideSpeed - ladder.normalZ * 0.45 * wishMoveFwd;
  state.vy = climbVy;
  state.isGrounded = true;

  const hRes = resolveHorizontalAABB(
    state.x,
    state.y,
    state.z,
    state.vx,
    state.vz,
    dt,
    KINEMATICS.CAPSULE_RADIUS,
    KINEMATICS.CAPSULE_HEIGHT_STAND,
    colliders,
    true,
    KINEMATICS.STEP_HEIGHT,
    bounds
  );
  state.x = hRes.x;
  state.z = hRes.z;
  state.y = Math.max(ladder.minY, Math.min(ladder.maxY + 0.05, hRes.y + state.vy * dt));
  return true;
}

/**
 * Deterministic CS 1.6 multi-pellet buckshot dispersion (for Leone M3 = 9 pellets, XM1014 = 6 pellets).
 * Pellet 0 is tight center-biased while outer pellets trace a golden-angle ring pattern.
 */
export function executeShotgunRaycast(
  origin: Vec3,
  yaw: number,
  pitch: number,
  pelletCount: number,
  pelletSpreadRad: number,
  baseDamagePerPellet: number,
  armorPenetration: number,
  penetrationPower: number,
  blocks: MapBlockCollider[],
  targets: TargetCollider[],
  rangeDropoff: number,
  spatialGrid?: SpatialGrid,
  headDamageMultiplier = 4.0,
  seed = 1
): PelletRayOutcome[] {
  const outcomes: PelletRayOutcome[] = [];
  const goldenAngle = 2.399963229728653;

  for (let p = 0; p < pelletCount; p++) {
    let sx = 0;
    let sy = 0;
    if (p > 0) {
      const hash = Math.abs(Math.sin((seed + p) * 19.371) * 43758.5453) % 1;
      const radial = Math.sqrt(p / pelletCount) * pelletSpreadRad * (0.78 + hash * 0.44);
      const theta = p * goldenAngle + hash * 0.55;
      sx = Math.cos(theta) * radial;
      sy = Math.sin(theta) * radial;
    }
    const dir = computeSpreadRayDirection(yaw, pitch, sx, sy);
    const hits = executeTacticalRaycast(
      { origin, direction: dir },
      baseDamagePerPellet,
      armorPenetration,
      penetrationPower,
      75, // Effective shotgun range
      blocks,
      targets,
      rangeDropoff,
      spatialGrid,
      headDamageMultiplier
    );
    outcomes.push({ rayDirection: dir, hits });
  }
  return outcomes;
}

/**
 * Checks whether a target holding a raised Tactical Shield blocks an incoming bullet vector.
 */
export function isBlockedByTacticalShield(
  bulletDir: Vec3,
  targetYaw: number,
  hasRaisedShield: boolean,
  hitZone?: HitZone
): boolean {
  if (!hasRaisedShield) return false;
  if (hitZone === 'LEGS') return false; // Feet remain exposed under raised shield (authentic CS 1.6 counterplay!)
  // Target forward (-Z in Three.js yaw convention)
  const tgtFwdX = -Math.sin(targetYaw);
  const tgtFwdZ = -Math.cos(targetYaw);
  const incomingTowardShooterX = -bulletDir.x;
  const incomingTowardShooterZ = -bulletDir.z;
  const len = Math.hypot(incomingTowardShooterX, incomingTowardShooterZ);
  if (len < 1e-6) return false;
  const dot = (tgtFwdX * incomingTowardShooterX + tgtFwdZ * incomingTowardShooterZ) / len;
  return dot >= 0.42; // ~65 deg frontal ballistic shield arc
}

        else bestMat = b.material;
      }

  prefix: string,
  axis: 'x' | 'z',
  startCoord: number,
  endCoord: number,
  crossMin: number,
  crossMax: number,
  startY: number,
  endY: number,

  const steps: MapBlockCollider[] = [];
  const totalSpan = endCoord - startCoord;
  const totalRise = endY - startY;

  const risePerStep = totalRise / stepCount;

      maxX: axis === 'x' ? Math.max(c0, c1) : crossMax,

      maxZ: axis === 'z' ? Math.max(c0, c1) : crossMax,

}

function buildStairs(
  prefix: string,
  axis: 'x' | 'z',
  startCoord: number,
  endCoord: number,
  crossMin: number,
  crossMax: number,
  startY: number,
  endY: number,
  stepCount: number,
  theme: MapBlockCollider['theme'] = 'sanctum'
): MapBlockCollider[] {
  const steps: MapBlockCollider[] = [];
  const run = (endCoord - startCoord) / stepCount;
  const rise = (endY - startY) / stepCount;
  for (let i = 0; i < stepCount; i++) {
    const c0 = startCoord + i * run;
    const c1 = startCoord + (i + 1) * run;
    steps.push({
      id: `${prefix}_${i + 1}`,
      minX: axis === 'x' ? Math.min(c0, c1) : crossMin,
      maxX: axis === 'x' ? Math.max(c0, c1) : crossMax,
      minY: Number((startY + i * rise).toFixed(3)),
      maxY: Number((startY + (i + 1) * rise).toFixed(3)),
      minZ: axis === 'z' ? Math.min(c0, c1) : crossMin,
      maxZ: axis === 'z' ? Math.max(c0, c1) : crossMax,
      material: 'CONCRETE',
      theme,
    });
  }
  return steps;
}

export type HostageFollowState = 'IDLE' | 'FOLLOWING' | 'RESCUED' | 'DEAD';

export interface HostageEntityState {
  id: string;
  name: string;
  pos: Vec3;
  yaw: number;
  health: number;
  state: HostageFollowState;
  escortId: string | null;
}

export function isPointInRescueZone(pos: Vec3, zones: readonly RescueZoneSpec[]): boolean {
  for (let i = 0; i < zones.length; i++) {
    const z = zones[i];
    if (pos.x >= z.minX && pos.x <= z.maxX && pos.z >= z.minZ && pos.z <= z.maxZ) {
      return true;
    }
  }
  return false;
}

/**
 * Pure deterministic CS 1.6 Hostage Rescue rule evaluator:
 * - Rescuing >50% of alive hostages (or all 4) triggers Vanguard CT win.
 * - Time expiration gives Syndicate T win (hostages not rescued).
 */
export function evaluateHostageWinCondition(hostages: readonly HostageEntityState[]): {
  isRoundOver: boolean;
  winner?: 'Vanguard' | 'Syndicate';
  reason?: string;
} {
  let rescued = 0;
  let alive = 0;
  for (const h of hostages) {
    if (h.state === 'RESCUED') rescued++;
    else if (h.state !== 'DEAD') alive++;
  }
  if (rescued > 0 && alive === 0) {
    return { isRoundOver: true, winner: 'Vanguard', reason: 'All Hostages Rescued' };
  }
  // Classic CS 1.6 rule: rescuing majority (>= 3 of 4) wins if remaining are dead/unreachable, or all rescued
  if (rescued >= Math.ceil(hostages.length * 0.75) && rescued >= 3) {
    return { isRoundOver: true, winner: 'Vanguard', reason: 'Hostages Rescued!' };
  }
  return { isRoundOver: false };
}

/**
 * Computes deterministic ordered purchase item IDs for [F1] AutoBuy.
 */
export function computeAutoBuySequence(ctx: BuyPlanContext): string[] {
  const plan: string[] = [];
  let cashLeft = ctx.cash;
  let hasPrimary = Boolean(ctx.currentPrimaryId);
  let armor = ctx.armor;
  let hasHelmet = ctx.hasHelmet;
  let hasKit = ctx.hasDefuseKit;

  const list = ctx.faction === 'Vanguard' ? AUTOBUY_PRIORITY_VANGUARD : AUTOBUY_PRIORITY_SYNDICATE;

  for (const token of list) {
    if (token === 'primammo' || token === 'secammo') {
      plan.push(token);
      continue;
    }
    if (token === 'defuse-kit') {
      if (ctx.faction === 'Vanguard' && !hasKit && cashLeft >= ECONOMY.prices.defuseKit) {
        plan.push('defuse-kit');
        cashLeft -= ECONOMY.prices.defuseKit;
        hasKit = true;
      }
      continue;
    }
    if (token === 'kevlar-helmet') {
      if (!(armor >= 100 && hasHelmet) && cashLeft >= ECONOMY.prices.kevlarHelmetCombo) {
        plan.push('kevlar-helmet');
        cashLeft -= ECONOMY.prices.kevlarHelmetCombo;
        armor = 100;
        hasHelmet = true;
      }
      continue;
    }
    if (token === 'kevlar') {
      if (armor < 100 && cashLeft >= ECONOMY.prices.kevlar) {
        plan.push('kevlar');
        cashLeft -= ECONOMY.prices.kevlar;
        armor = 100;
      }
      continue;
    }
    const def = WEAPONS[token];
    if (def && def.slot === 1 && !hasPrimary && cashLeft >= def.price) {
      plan.push(def.id);
      cashLeft -= def.price;
      hasPrimary = true;
    }
  }
  return plan;
}

/**
 * Computes deterministic ordered purchase item IDs for [F4] Rebuy.
 */
export function computeRebuySequence(ctx: BuyPlanContext, snap: RebuySnapshot): string[] {
  const plan: string[] = [];
  let cashLeft = ctx.cash;
  let armor = ctx.armor;
  let hasHelmet = ctx.hasHelmet;
  let hasKit = ctx.hasDefuseKit;
  let hasNvg = ctx.hasNightVision;

  for (const cat of DEFAULT_REBUY_ORDER) {
    if (cat === 'PrimaryWeapon' && snap.primaryWeaponId && ctx.currentPrimaryId !== snap.primaryWeaponId) {
      const def = WEAPONS[snap.primaryWeaponId];
      if (def && cashLeft >= def.price) {
        plan.push(def.id);
        cashLeft -= def.price;
      }
    } else if (cat === 'PrimaryAmmo') {
      plan.push('primammo');
    } else if (cat === 'SecondaryWeapon' && snap.secondaryWeaponId && ctx.currentSecondaryId !== snap.secondaryWeaponId) {
      const def = WEAPONS[snap.secondaryWeaponId];
      if (def && cashLeft >= def.price) {
        plan.push(def.id);
        cashLeft -= def.price;
      }
    } else if (cat === 'SecondaryAmmo') {
      plan.push('secammo');
    } else if (cat === 'Defuser' && ctx.faction === 'Vanguard' && snap.hasDefuseKit && !hasKit) {
      if (cashLeft >= ECONOMY.prices.defuseKit) {
        plan.push('defuse-kit');
        cashLeft -= ECONOMY.prices.defuseKit;
        hasKit = true;
      }
    } else if (cat === 'Armor' && snap.armorTier !== 'none') {
      if (snap.armorTier === 'kevlar-helmet' && !(armor >= 100 && hasHelmet) && cashLeft >= 1000) {
        plan.push('kevlar-helmet');
        cashLeft -= 1000;
        armor = 100;
        hasHelmet = true;
      } else if (armor < 100 && cashLeft >= 650) {
        plan.push('kevlar');
        cashLeft -= 650;
        armor = 100;
      }
    } else if (cat === 'HEGrenade' && snap.utilities.includes('he-grenade') && (ctx.ownedUtilities['he-grenade'] || 0) < 1 && cashLeft >= 300) {
      plan.push('he-grenade');
      cashLeft -= 300;
    } else if (cat === 'Flashbang' && snap.utilities.includes('flashbang') && (ctx.ownedUtilities['flashbang'] || 0) < 2 && cashLeft >= 200) {
      plan.push('flashbang');
      cashLeft -= 200;
    } else if (cat === 'SmokeGrenade' && snap.utilities.includes('smoke-grenade') && (ctx.ownedUtilities['smoke-grenade'] || 0) < 1 && cashLeft >= 300) {
      plan.push('smoke-grenade');
      cashLeft -= 300;
    } else if (cat === 'Molotov' && snap.utilities.includes('molotov') && (ctx.ownedUtilities['molotov'] || 0) < 1 && cashLeft >= 400) {
      plan.push('molotov');
      cashLeft -= 400;
    } else if (cat === 'NightVision' && snap.hasNightVision && !hasNvg && cashLeft >= 1250) {
      plan.push('nvg');
      cashLeft -= 1250;
      hasNvg = true;
    }
  }
  return plan;
}

import * as THREE from 'three';

/**
 * Clean-room articulated 3D Hostage character rig for CS_OFFICE Hostage Rescue mode.
 * Distinct silhouette from combat operators (tan trench coat / white dress shirt + tie)
 * with idle nervous sway and following jog gait.
 */
export class HostageRig {
  public readonly root = new THREE.Group();
  private readonly torso = new THREE.Group();
  private readonly leftArm = new THREE.Group();
  private readonly rightArm = new THREE.Group();
  private readonly leftLeg = new THREE.Group();
  private readonly rightLeg = new THREE.Group();
  private readonly statusRing: THREE.Mesh;
  private animPhase = Math.random() * Math.PI * 2;

  constructor(name: string) {
    this.root.name = `hostage_${name}`;

    const coatMat = new THREE.MeshStandardMaterial({ color: 0x9c8970, roughness: 0.78, metalness: 0.05 });
    const shirtMat = new THREE.MeshStandardMaterial({ color: 0xe5e7eb, roughness: 0.72, metalness: 0.02 });
    const tieMat = new THREE.MeshStandardMaterial({ color: 0x991b1b, roughness: 0.55, metalness: 0.1 });
    const pantsMat = new THREE.MeshStandardMaterial({ color: 0x262a30, roughness: 0.82, metalness: 0.05 });
    const skinMat = new THREE.MeshStandardMaterial({ color: 0xdab294, roughness: 0.68, metalness: 0.02 });
    const hairMat = new THREE.MeshStandardMaterial({ color: 0x2a221d, roughness: 0.85, metalness: 0.02 });

    // Legs (hip at y = 0.86m)
    const makeLeg = (side: -1 | 1, pivot: THREE.Group) => {
      pivot.position.set(side * 0.10, 0.86, 0);
      const legMesh = new THREE.Mesh(new THREE.BoxGeometry(0.14, 0.84, 0.15), pantsMat);
      legMesh.position.y = -0.42;
      legMesh.castShadow = true;
      pivot.add(legMesh);
      this.root.add(pivot);
    };
    makeLeg(-1, this.leftLeg);
    makeLeg(1, this.rightLeg);

    // Torso (y = 0.86 -> 1.48m)
    this.torso.position.set(0, 0.86, 0);
    const jacket = new THREE.Mesh(new THREE.BoxGeometry(0.38, 0.62, 0.22), coatMat);
    jacket.position.y = 0.31;
    jacket.castShadow = true;
    const shirt = new THREE.Mesh(new THREE.BoxGeometry(0.14, 0.45, 0.23), shirtMat);
    shirt.position.set(0, 0.36, 0.01);
    const tie = new THREE.Mesh(new THREE.BoxGeometry(0.04, 0.32, 0.24), tieMat);
    tie.position.set(0, 0.34, 0.015);
    const head = new THREE.Mesh(new THREE.BoxGeometry(0.22, 0.24, 0.22), skinMat);
    head.position.set(0, 0.76, 0);
    head.castShadow = true;
    const hair = new THREE.Mesh(new THREE.BoxGeometry(0.23, 0.08, 0.23), hairMat);
    hair.position.set(0, 0.86, -0.01);
    this.torso.add(jacket, shirt, tie, head, hair);

    const makeArm = (side: -1 | 1, pivot: THREE.Group) => {
      pivot.position.set(side * 0.24, 0.56, 0);
      const sleeve = new THREE.Mesh(new THREE.BoxGeometry(0.11, 0.52, 0.11), coatMat);
      sleeve.position.y = -0.24;
      sleeve.castShadow = true;
      const hand = new THREE.Mesh(new THREE.BoxGeometry(0.09, 0.10, 0.09), skinMat);
      hand.position.y = -0.52;
      pivot.add(sleeve, hand);
      this.torso.add(pivot);
    };
    makeArm(-1, this.leftArm);
    makeArm(1, this.rightArm);

    this.root.add(this.torso);

    // Subtle ground indicator ring (amber = waiting, emerald = following)
    const ringGeo = new THREE.RingGeometry(0.42, 0.50, 24);
    ringGeo.rotateX(-Math.PI / 2);
    const ringMat = new THREE.MeshBasicMaterial({ color: 0xf59e0b, side: THREE.DoubleSide, transparent: true, opacity: 0.72 });
    this.statusRing = new THREE.Mesh(ringGeo, ringMat);
    this.statusRing.position.y = 0.03;
    this.root.add(this.statusRing);
  }

  public update(dt: number, isFollowing: boolean, moveSpeed: number): void {
    this.animPhase += dt * (moveSpeed > 0.3 ? 10.5 : 2.2);
    const mat = this.statusRing.material as THREE.MeshBasicMaterial;
    mat.color.setHex(isFollowing ? 0x10b981 : 0xf59e0b);

    if (moveSpeed > 0.3) {
      const swing = Math.sin(this.animPhase) * 0.55;
      this.leftLeg.rotation.x = swing;
      this.rightLeg.rotation.x = -swing;
      this.leftArm.rotation.x = -swing * 0.75;
      this.rightArm.rotation.x = swing * 0.75;
      this.torso.position.y = 0.86 + Math.abs(Math.cos(this.animPhase)) * 0.03;
    } else {
      // Surrendered/nervous stance
      this.leftLeg.rotation.x = 0;
      this.rightLeg.rotation.x = 0;
      this.leftArm.rotation.x = -0.35 + Math.sin(this.animPhase) * 0.05;
      this.rightArm.rotation.x = -0.35 - Math.sin(this.animPhase) * 0.05;
      this.torso.position.y = 0.86;
    }
  }

  public dispose(): void {
    this.root.traverse((obj) => {
      if ((obj as THREE.Mesh).isMesh) {
        const m = obj as THREE.Mesh;
        m.geometry?.dispose();
        if (Array.isArray(m.material)) m.material.forEach((x) => x.dispose());
        else m.material?.dispose();
      }
    });
  }
}

    });

import * as THREE from 'three';
import { AABBCollider, BreakableBlockSpec } from '@garam/shared';
import { ProceduralAudio } from '../audio/ProceduralAudio.js';
import { EffectsManager } from '../combat/EffectsManager.js';

interface BreakableRuntime {
  spec: BreakableBlockSpec;
  hp: number;
  broken: boolean;
  mesh: THREE.Mesh;
  colliderRef: AABBCollider;
}

/**
 * Manages CS 1.6 func_breakable glass panes & air vents across maps:
 * - Shatters when shot/knifed, disabling the matching collider so players/bullets pass through!
 * - Restores all glass/vents cleanly on round reset.
 */
export class BreakablePropManager {
  private entries: BreakableRuntime[] = [];
  private scene: THREE.Scene;
  private audio: ProceduralAudio;
  private effects: EffectsManager;

  constructor(scene: THREE.Scene, audio: ProceduralAudio, effects: EffectsManager) {
    this.scene = scene;
    this.audio = audio;
    this.effects = effects;
  }

  public initForMap(specs: readonly BreakableBlockSpec[], mapColliders: AABBCollider[]): void {
    this.dispose();
    const glassMat = new THREE.MeshPhysicalMaterial({
      color: 0xaad8ff,
      roughness: 0.08,
      metalness: 0.12,
      transparent: true,
      opacity: 0.45,
      side: THREE.DoubleSide,
    });
    const ventMat = new THREE.MeshStandardMaterial({
      color: 0x5c6670,
      roughness: 0.42,
      metalness: 0.75,
    });

    for (const spec of specs) {
      const w = Math.max(0.05, spec.maxX - spec.minX);
      const h = Math.max(0.05, spec.maxY - spec.minY);
      const d = Math.max(0.05, spec.maxZ - spec.minZ);
      const mesh = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), spec.label === 'GLASS' ? glassMat : ventMat);
      mesh.position.set((spec.minX + spec.maxX) * 0.5, (spec.minY + spec.maxY) * 0.5, (spec.minZ + spec.maxZ) * 0.5);
      mesh.userData.isMapRoot = true;
      mesh.userData.breakableId = spec.id;
      this.scene.add(mesh);

      const col: AABBCollider = {
        minX: spec.minX, maxX: spec.maxX,
        minY: spec.minY, maxY: spec.maxY,
        minZ: spec.minZ, maxZ: spec.maxZ,
      };
      mapColliders.push(col);

      this.entries.push({ spec, hp: spec.health, broken: false, mesh, colliderRef: col });
    }
  }

  public applyDamage(blockId: string, damage: number): boolean {
    const entry = this.entries.find((e) => e.spec.id === blockId && !e.broken);
    if (!entry) return false;
    entry.hp -= damage;
    if (entry.hp <= 0) {
      this.shatter(entry);
      return true;
    }
    return false;
  }

  public checkRayHit(origin: THREE.Vector3, dir: THREE.Vector3, maxDist = 40): boolean {
    let bestT = maxDist;
    let bestEntry: BreakableRuntime | null = null;
    const box = new THREE.Box3();
    const ray = new THREE.Ray(origin, dir);
    const hitPt = new THREE.Vector3();

    for (const e of this.entries) {
      if (e.broken) continue;
      box.min.set(e.spec.minX, e.spec.minY, e.spec.minZ);
      box.max.set(e.spec.maxX, e.spec.maxY, e.spec.maxZ);
      if (ray.intersectBox(box, hitPt)) {
        const d = origin.distanceTo(hitPt);
        if (d < bestT) {
          bestT = d;
          bestEntry = e;
        }
      }
    }
    if (bestEntry) {
      this.shatter(bestEntry);
      return true;
    }
    return false;
  }

  private shatter(entry: BreakableRuntime): void {
    entry.broken = true;
    entry.mesh.visible = false;
    // Move collider safely out of bounds without mutating array indices
    entry.colliderRef.minY = -999;
    entry.colliderRef.maxY = -998;
    const center = entry.mesh.position;
    this.effects.spawnImpactBurst(center, new THREE.Vector3(0, 1, 0), entry.spec.label === 'GLASS' ? 'METAL_SHEET' : 'METAL_SHEET');
    if (entry.spec.label === 'GLASS') {
      this.audio.playGlassShatter?.(center);
    } else {
      this.audio.playSurfaceImpact('METAL_SHEET', center);
    }
  }

  public resetRound(): void {
    for (const e of this.entries) {
      e.broken = false;
      e.hp = e.spec.health;
      e.mesh.visible = true;
      e.colliderRef.minY = e.spec.minY;
      e.colliderRef.maxY = e.spec.maxY;
    }
  }

  public dispose(): void {
    for (const e of this.entries) {
      this.scene.remove(e.mesh);
      e.mesh.geometry.dispose();
    }
    this.entries = [];
  }
}

function createNukeHazardTex(): THREE.CanvasTexture {
  const c = document.createElement('canvas');
  c.width = 256; c.height = 256;
  const ctx = c.getContext('2d')!;
  ctx.fillStyle = '#525b64';
  ctx.fillRect(0, 0, 256, 256);
  ctx.fillStyle = '#eab308';
  ctx.fillRect(0, 210, 256, 28);
  ctx.fillStyle = '#111827';
  for (let x = -32; x < 288; x += 36) {
    ctx.beginPath();
    ctx.moveTo(x, 210); ctx.lineTo(x + 18, 210);
    ctx.lineTo(x + 6, 238); ctx.lineTo(x - 12, 238);
    ctx.closePath(); ctx.fill();
  }
  const tex = new THREE.CanvasTexture(c);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  return tex;
}

export function buildDeNukeMap(scene: THREE.Scene): MapBuildResult {
  const root = new THREE.Group();
  root.name = 'de_nuke_map_root';
  root.userData.isMapRoot = true;

  const mats: Record<string, THREE.MeshStandardMaterial> = {
    concrete: new THREE.MeshStandardMaterial({ map: createNukeHazardTex(), roughness: 0.85, metalness: 0.08 }),
    foundry: new THREE.MeshStandardMaterial({ color: 0x4b5563, roughness: 0.72, metalness: 0.16 }),
    metal: new THREE.MeshStandardMaterial({ color: 0x3b6e8c, roughness: 0.36, metalness: 0.74 }),
    wood: new THREE.MeshStandardMaterial({ color: 0x8b5a2b, roughness: 0.78, metalness: 0.05 }),
  };

  // Ground Plane (Basement B-Site Floor y = 0.0m)
  const floor = new THREE.Mesh(new THREE.PlaneGeometry(110, 90), mats.foundry);
  floor.rotation.x = -Math.PI / 2;
  floor.receiveShadow = true;
  root.add(floor);

  for (const b of DE_NUKE_BLOCKS as MapBlockCollider[]) {
    const w = b.maxX - b.minX;
    const h = b.maxY - b.minY;
    const d = b.maxZ - b.minZ;
    const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), mats[b.theme || 'concrete'] || mats.concrete);
    m.position.set((b.minX + b.maxX) * 0.5, (b.minY + b.maxY) * 0.5, (b.minZ + b.maxZ) * 0.5);
    m.castShadow = true;
    m.receiveShadow = true;
    m.matrixAutoUpdate = false;
    m.updateMatrix();
    root.add(m);
  }

  // Visible 3D Steel Ladder Rungs for every LadderVolume
  const rungMat = new THREE.MeshStandardMaterial({ color: 0xeab308, roughness: 0.3, metalness: 0.85 });
  for (const l of DE_NUKE_LADDERS) {
    const height = l.maxY - l.minY;
    const cx = (l.minX + l.maxX) * 0.5;
    const cz = (l.minZ + l.maxZ) * 0.5;
    for (let y = l.minY + 0.35; y < l.maxY; y += 0.38) {

      rung.rotation.z = Math.abs(l.normalZ) > 0.5 ? Math.PI / 2 : 0;

    }
    // Side rails
    const rail = new THREE.Mesh(new THREE.BoxGeometry(0.05, height, 0.05), rungMat);
    rail.position.set(cx, l.minY + height * 0.5, cz);
    root.add(rail);

    const mesh = new THREE.Mesh(
      new THREE.PlaneGeometry(3.6, 3.6),
      new THREE.MeshBasicMaterial({ map: new THREE.CanvasTexture(c), transparent: true, depthWrite: false })
    );
    mesh.rotation.x = -Math.PI / 2;
    mesh.position.set(x, y, z);
    root.add(mesh);
  };
  addStencil('A', DE_NUKE_BOMBSITES.A.center.x, DE_NUKE_BOMBSITES.A.center.y + 0.04, DE_NUKE_BOMBSITES.A.center.z, '#ef4444');
  addStencil('B', DE_NUKE_BOMBSITES.B.center.x, DE_NUKE_BOMBSITES.B.center.y + 0.04, DE_NUKE_BOMBSITES.B.center.z, '#f59e0b');

  // Basement B Reactor Glow Light
  const bLight = new THREE.PointLight(0x38bdf8, 2.2, 28);
  bLight.position.set(4, 2.8, -5);
  root.add(bLight);

  scene.add(root);

  const bots: BotEntity[] = DE_NUKE_BOTS.map((cfg) => {
    const rig = new OperatorRig(cfg.team, cfg.name);
    rig.root.position.set(cfg.waypoints[0].x, cfg.waypoints[0].y, cfg.waypoints[0].z);

import * as THREE from 'three';

export class ProceduralAudioCS16 {
  public static playWeaponSynth(
    ctx: AudioContext,
    dest: AudioNode,
    noiseBuffer: AudioBuffer,
    weaponId: string,
    isSilenced = true
  ): boolean {
    const t = ctx.currentTime;
    const playNoiseBurst = (dur: number, freq: number, q: number, gainPeak: number, type: BiquadFilterType = 'bandpass') => {
      const src = ctx.createBufferSource();
      src.buffer = noiseBuffer;
      const filter = ctx.createBiquadFilter();
      filter.type = type;
      filter.frequency.setValueAtTime(freq, t);
      filter.Q.setValueAtTime(q, t);
      const gain = ctx.createGain();
      gain.gain.setValueAtTime(0.001, t);
      gain.gain.linearRampToValueAtTime(gainPeak, t + 0.004);
      gain.gain.exponentialRampToValueAtTime(0.001, t + dur);
      src.connect(filter);
      filter.connect(gain);
      gain.connect(dest);
      src.start(t);
      src.stop(t + dur + 0.02);
    };

    const playPunch = (startHz: number, endHz: number, dur: number, vol: number, wave: OscillatorType = 'triangle') => {
      const osc = ctx.createOscillator();
      osc.type = wave;
      osc.frequency.setValueAtTime(startHz, t);
      osc.frequency.exponentialRampToValueAtTime(endHz, t + dur);
      const gain = ctx.createGain();
      gain.gain.setValueAtTime(vol, t);
      gain.gain.exponentialRampToValueAtTime(0.001, t + dur);
      osc.connect(gain);
      gain.connect(dest);
      osc.start(t);
      osc.stop(t + dur + 0.02);
    };

    switch (weaponId) {
      case 'mp5-navy':
        playPunch(210, 58, 0.08, 0.42);
        playNoiseBurst(0.11, 1850, 1.4, 0.48);
        return true;
      case 'p90':
        playPunch(245, 72, 0.065, 0.36);
        playNoiseBurst(0.09, 2400, 1.8, 0.42);
        return true;
      case 'clarion-famas':
        playPunch(175, 48, 0.11, 0.48, 'sawtooth');
        playNoiseBurst(0.16, 1650, 1.1, 0.56);
        return true;
      case 'defender-galil':
        playPunch(155, 44, 0.13, 0.52, 'sawtooth');
        playNoiseBurst(0.19, 1380, 0.9, 0.62);
        return true;
      case 'schmidt-scout':
        playPunch(320, 42, 0.22, 0.68, 'sawtooth');
        playNoiseBurst(0.32, 2100, 0.7, 0.78, 'lowpass');
        return true;
      case 'leone-m3':
        playPunch(125, 28, 0.26, 0.85, 'sine');
        playNoiseBurst(0.36, 980, 0.55, 0.92, 'lowpass');
        // Signature pump-action rack clack at +0.26s
        setTimeout(() => {
          if (ctx.state === 'running') {
            const nt = ctx.currentTime;
            const o = ctx.createOscillator();
            o.type = 'square';
            o.frequency.setValueAtTime(520, nt);
            o.frequency.setValueAtTime(760, nt + 0.07);
            const g = ctx.createGain();
            g.gain.setValueAtTime(0.14, nt);
            g.gain.exponentialRampToValueAtTime(0.001, nt + 0.14);
            o.connect(g); g.connect(dest);
            o.start(nt); o.stop(nt + 0.15);
          }
        }, 260);
        return true;
      case 'xm1014':
        playPunch(135, 32, 0.20, 0.78, 'sine');
        playNoiseBurst(0.28, 1120, 0.6, 0.84, 'lowpass');
        return true;
      case 'dual-elites':
        playPunch(260, 62, 0.09, 0.44);
        playNoiseBurst(0.12, 2100, 1.2, 0.50);
        return true;
      case 'vanguard-m4':
        if (!isSilenced) {
          playPunch(185, 45, 0.14, 0.58, 'sawtooth');
          playNoiseBurst(0.22, 1650, 0.85, 0.72);
          return true;
        }
        return false;
      case 'usp-s':
        if (!isSilenced) {
          playPunch(240, 56, 0.10, 0.50);
          playNoiseBurst(0.14, 1950, 1.0, 0.58);
          return true;
        }
        return false;

import { MapBlockCollider, SpatialGrid, executeTacticalRaycast } from '@garam/shared';
import { ProceduralAudio } from '../audio/ProceduralAudio.js';
import { HUDManager } from '../ui/HUDManager.js';

export type SprayEmblemId = 'sovereign' | 'target' | 'biohazard' | 'crown' | 'spade' | 'vanguard';

export class SprayManager {
  private scene: THREE.Scene;
  private audio: ProceduralAudio;
  private hud: HUDManager;
  private activeDecals: THREE.Mesh[] = [];
  private readonly MAX_DECALS = 16;
  private lastSprayMs = -99999;
  private readonly COOLDOWN_MS = 20000;
  private textures = new Map<SprayEmblemId, THREE.CanvasTexture>();

  constructor(scene: THREE.Scene, audio: ProceduralAudio, hud: HUDManager) {
    this.scene = scene;
    this.audio = audio;
    this.hud = hud;
  }

  private getSprayTexture(id: SprayEmblemId): THREE.CanvasTexture {
    const existing = this.textures.get(id);
    if (existing) return existing;

    const c = document.createElement('canvas');
    c.width = 256; c.height = 256;
    const ctx = c.getContext('2d')!;
    ctx.clearRect(0, 0, 256, 256);

    const accent =
      id === 'biohazard' ? '#eab308' :
      id === 'vanguard' ? '#3b82f6' :
      id === 'target' ? '#ef4444' :
      id === 'crown' ? '#f59e0b' : '#f43f5e';

    ctx.strokeStyle = accent;
    ctx.fillStyle = accent;
    ctx.lineWidth = 10;

    ctx.beginPath();
    ctx.arc(128, 128, 102, 0, Math.PI * 2);
    ctx.stroke();

    ctx.font = '900 92px sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    const glyph =
      id === 'biohazard' ? '☣' :
      id === 'target' ? '⌖' :
      id === 'crown' ? '♛' :
      id === 'spade' ? '♠' :
      id === 'vanguard' ? '❖' : 'G';
    ctx.fillText(glyph, 128, 122);

    ctx.font = '800 22px sans-serif';
    ctx.fillStyle = '#ffffff';
    ctx.fillText(id.toUpperCase(), 128, 202);

    const tex = new THREE.CanvasTexture(c);
    tex.colorSpace = THREE.SRGBColorSpace;
    this.textures.set(id, tex);
    return tex;
  }

  public trySpray(
    camera: THREE.Camera,
    blocks: MapBlockCollider[],
    spatialGrid?: SpatialGrid,
    emblemId: SprayEmblemId = 'sovereign'
  ): { pos: THREE.Vector3; normal: THREE.Vector3 } | null {
    const now = performance.now();
    if (now - this.lastSprayMs < this.COOLDOWN_MS) {
      const rem = Math.ceil((this.COOLDOWN_MS - (now - this.lastSprayMs)) / 1000);
      this.hud.addRadioMessage(`Spray recharge: ${rem}s remaining`);
      return null;
    }

    const dir = new THREE.Vector3();
    camera.getWorldDirection(dir);
    const hits = executeTacticalRaycast(
      {
        origin: { x: camera.position.x, y: camera.position.y, z: camera.position.z },
        direction: { x: dir.x, y: dir.y, z: dir.z },
      },
      1, 0, 0, 2.8, blocks, [], 1.0, spatialGrid
    );
    if (hits.length === 0 || hits[0].targetId) return null;

    const h = hits[0];
    const pos = new THREE.Vector3(h.point.x, h.point.y, h.point.z);
    const norm = new THREE.Vector3(h.normal.x, h.normal.y, h.normal.z).normalize();
    this.spawnSprayAt(emblemId, pos, norm);
    this.lastSprayMs = now;
    this.audio.playSprayAerosol?.();
    return { pos, normal: norm };
  }

  public spawnSprayAt(emblemId: SprayEmblemId, pos: THREE.Vector3, normal: THREE.Vector3): void {
    const mat = new THREE.MeshBasicMaterial({
      map: this.getSprayTexture(emblemId),
      transparent: true,
      opacity: 0.92,
      depthWrite: false,
      polygonOffset: true,
      polygonOffsetFactor: -2,
    });
    const mesh = new THREE.Mesh(new THREE.PlaneGeometry(0.95, 0.95), mat);
    mesh.position.copy(pos).addScaledVector(normal, 0.018);
    mesh.lookAt(pos.clone().add(normal));
    this.scene.add(mesh);
    this.activeDecals.push(mesh);

    if (this.activeDecals.length > this.MAX_DECALS) {
      const oldest = this.activeDecals.shift();
      if (oldest) {
        this.scene.remove(oldest);
        oldest.geometry.dispose();
        (oldest.material as THREE.Material).dispose();
      }
    }
  }

  public clearAll(): void {
    for (const m of this.activeDecals) {
      this.scene.remove(m);
      m.geometry.dispose();
      (m.material as THREE.Material).dispose();
    }
    this.activeDecals = [];
  }
}

      m.position.set((rz.minX + rz.maxX) * 0.5, 0.04, (rz.minZ + rz.maxZ) * 0.5);
      this.scene.add(m);
      this.rescueZoneMesh = m;
    }
  }

  public tryInteract(
    actorId: string,
    actorTeam: 'Vanguard' | 'Syndicate',
    actorPos: THREE.Vector3,
    onCashAward: (amount: number) => void
  ): boolean {
    if (!this.enabled || actorTeam !== 'Vanguard') return false;
    for (const h of this.hostages) {
      if (h.state === 'RESCUED' || h.state === 'DEAD') continue;
      const d = Math.hypot(h.pos.x - actorPos.x, h.pos.z - actorPos.z);
      if (d <= 2.4 && Math.abs(h.pos.y - actorPos.y) < 1.8) {
        if (h.state === 'IDLE') {
          h.state = 'FOLLOWING';
          h.escortId = actorId;
          onCashAward(150);
          this.hud.addRadioMessage(`${h.name}: "Okay, let's get out of here!" (+$150)`);
          this.audio.playRadioClick?.('Okay, let us get out of here!', 1.05);
        } else {
          h.state = 'IDLE';
          h.escortId = null;
          this.hud.addRadioMessage(`${h.name}: "Wait right here? Okay."`);
        }
        return true;
      }
    }
    return false;
  }

  public update(
    dt: number,
    playerFeet: THREE.Vector3,
    playerTeam: 'Vanguard' | 'Syndicate',
    colliders: AABBCollider[],
    bounds: MapBounds,
    onCashAward: (amount: number) => void,
    onAllRescued: (reason: string) => void
  ): void {
    if (!this.enabled) return;

    for (const h of this.hostages) {
      if (h.state === 'RESCUED' || h.state === 'DEAD') {
        h.rig.root.visible = false;
        continue;
      }
      let moveSpeed = 0;
      if (h.state === 'FOLLOWING' && h.escortId === 'local' && playerTeam === 'Vanguard') {
        const dx = playerFeet.x - h.pos.x;
        const dz = playerFeet.z - h.pos.z;
        const dist = Math.hypot(dx, dz);
        if (dist > 1.85) {
          const speed = Math.min(4.6, dist * 2.4);
          moveSpeed = speed;
          const res = resolveHorizontalAABB(
            h.pos.x, h.pos.y, h.pos.z,
            (dx / dist) * speed, (dz / dist) * speed,
            dt, KINEMATICS.CAPSULE_RADIUS * 0.9, KINEMATICS.CAPSULE_HEIGHT_STAND,
            colliders, true, KINEMATICS.STEP_HEIGHT, bounds
          );
          h.pos.x = res.x; h.pos.y = res.y; h.pos.z = res.z;
          h.yaw = Math.atan2(dx, dz);
        }

        if (isPointInRescueZone(h.pos, this.rescueZones)) {
          h.state = 'RESCUED';
          h.escortId = null;
          h.rig.root.visible = false;
          onCashAward(1000);
          this.hud.showCenterBanner('HOSTAGE RESCUED!', `${h.name} reached safety (+$1,000)`, 2200);
          this.audio.playRadioClick?.('Hostage has been rescued!', 1.0);
          const win = evaluateHostageWinCondition(this.hostages);
          if (win.isRoundOver && win.reason) {
            onAllRescued(win.reason);
          }
        }
      }
      h.rig.root.position.set(h.pos.x, h.pos.y, h.pos.z);
      h.rig.root.rotation.y = h.yaw;
      h.rig.update(dt, h.state === 'FOLLOWING', moveSpeed);
    }
  }

  public resetRound(spawns: readonly HostageSpawnSpec[]): void {
    if (!this.enabled) return;
    for (let i = 0; i < this.hostages.length && i < spawns.length; i++) {
      const h = this.hostages[i];
      const s = spawns[i];
      h.pos = { ...s.pos };
      h.yaw = s.yaw;
      h.health = 100;
      h.state = 'IDLE';
      h.escortId = null;
      h.rig.root.visible = true;
      h.rig.root.position.set(s.pos.x, s.pos.y, s.pos.z);
    }
  }

  public dispose(): void {
    for (const h of this.hostages) {
      this.scene.remove(h.rig.root);
      h.rig.dispose();
    }
    this.hostages = [];
    if (this.rescueZoneMesh) {
      this.scene.remove(this.rescueZoneMesh);
      this.rescueZoneMesh.geometry.dispose();
      (this.rescueZoneMesh.material as THREE.Material).dispose();
      this.rescueZoneMesh = null;
    }
  }
}
